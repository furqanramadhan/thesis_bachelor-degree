\begin{abstracteng}

\textit{This research aims to implement the Machine Learning algorithm, K-Nearest Neighbors (KNN), to predict user quality on a web-based e-learning application. Using a dummy dataset consisting of features such as average access time, number of weekly logins, video duration watched, and user ratings, the KNN model was trained to classify users into categories: “Good”, “Fair”, and “Poor”. Evaluation results showed that the model achieved an accuracy of 87\% with k=5. This model can be integrated into recommendation systems to enhance the quality of e-learning services.}

\bigskip
\noindent
\textit{\textbf{Keywords :}} \textit{KNN, user quality prediction, e-learning, machine learning, web-based application}
\end{abstracteng}
